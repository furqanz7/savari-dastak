export type CafeScene = { setCounter(value: boolean): void; dispose(): void };
export function cafeCamera(counter: boolean, mobile: boolean) {
  return { position: counter ? [0, 1.6, 1.65] : [mobile ? 0 : -.45, 1.62, mobile ? -4.7 : -5.5],
    target: counter ? [0, 1.65, 4.22] : [0, 1.65, 3.5], fov: mobile ? (counter ? 1.0 : 1.18) : .94 };
}
